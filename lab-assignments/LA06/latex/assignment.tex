\section*{Objective}

\begin{itemize}[leftmargin=*]

  \item Turn free text into numeric features with bag-of-words and TF-IDF.

  \item Fit and interpret a Naive Bayes text classifier.

  \item Choose a metric appropriate to a task where one error type is far worse.

\end{itemize}

\section*{Datasets used in this assignment}

\textbf{C2 (SMS Spam Collection).}\\ Full descriptions, sources and loading snippets for all anchor datasets are in \texttt{Anchor-Datasets.pdf}, alongside this brief.

\section*{Instructions}

\begin{itemize}[leftmargin=*]

  \item Use a train/test split (80/20 unless stated) and set \texttt{random\_state} for reproducibility.

  \item Fit every transformer inside a \texttt{Pipeline} so nothing is fitted on the test set.

  \item Report the metric named in the question. A number without units or context earns no marks.

  \item Every question asks for a short written observation. Code without commentary caps you at half marks.

  \item The lab quiz that follows will ask you to read, trace and modify \emph{your own} submitted code.

\end{itemize}

\textbf{Submit:}

\begin{itemize}[leftmargin=*]

  \item Notebook: \texttt{AML\_LA06\_<SAPID>.ipynb} with all cells executed and outputs visible.

  \item Report: 2--3 pages PDF, or a \texttt{README.md}, carrying your results table and observations.

  \item Do not upload datasets. Cite the source and include the loading code.

\end{itemize}

\section*{Section A (Easy) --- attempt any 2 of 3 \quad (2 $\times$ 2 = 4)}

\begin{enumerate}[label=\textbf{A\arabic*.}, leftmargin=*]

  \item \textbf{Text to vectors} \QMarks{2}\\
    \emph{Dataset: C2.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Load the corpus and report the number of messages and the spam rate.
      \item Fit a \texttt{CountVectorizer} and report the vocabulary size.
      \item Show the ten most frequent tokens and state in two lines why they are useless for classification.
    \end{enumerate}

  \item \textbf{Naive Bayes} \QMarks{2}\\
    \emph{Dataset: C2.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Fit \texttt{MultinomialNB} on the counts and report accuracy, precision, recall and F1 on the spam class.
      \item Print the confusion matrix.
      \item State in two lines which words the model finds most spam-indicative.
    \end{enumerate}

  \item \textbf{Counts versus TF-IDF} \QMarks{2}\\
    \emph{Dataset: C2.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Refit the same classifier on \texttt{TfidfVectorizer} features.
      \item Report the same four metrics and compare with A2 in a small table.
      \item Explain in two lines what the IDF term is doing.
    \end{enumerate}

\end{enumerate}

\section*{Section B (Medium) --- attempt any 1 of 2 \quad (1 $\times$ 4 = 4)}

\begin{enumerate}[label=\textbf{B\arabic*.}, leftmargin=*]

  \item \textbf{The cost of a false positive} \QMarks{4}\\
    \emph{Dataset: C2.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item For your best model, report precision and recall on the spam class at thresholds 0.3, 0.5, 0.7 and 0.9.
      \item Plot the precision-recall curve and mark the four points.
      \item A real filter sends spam to a junk folder. Argue in 6--8 lines which error is worse and pick an operating threshold.
    \end{enumerate}

  \item \textbf{n-grams and vocabulary size} \QMarks{4}\\
    \emph{Dataset: C2.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Fit the same classifier with \texttt{ngram\_range} of (1,1), (1,2) and (1,3).
      \item Report vocabulary size, F1 and fit time for each in one table.
      \item Explain in 4--6 lines what bigrams add, and at what point the extra features stop paying for themselves.
    \end{enumerate}

\end{enumerate}
