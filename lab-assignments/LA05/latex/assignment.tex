\section*{Objective}

\begin{itemize}[leftmargin=*]

  \item Treat an image as a feature vector and classify it.

  \item Read a multiclass confusion matrix and find the confusable pairs.

  \item Compare a distance-based, a margin-based and a tree-based classifier on the same data.

\end{itemize}

\section*{Datasets used in this assignment}

\textbf{B1 (scikit-learn digits), A-small (Iris).}\\ Full descriptions, sources and loading snippets for all anchor datasets are in \texttt{Anchor-Datasets.pdf}, alongside this brief.

\section*{Instructions}

\begin{itemize}[leftmargin=*]

  \item Use a train/test split (80/20 unless stated) and set \texttt{random\_state} for reproducibility.

  \item Fit every transformer inside a \texttt{Pipeline} so nothing is fitted on the test set.

  \item Report the metric named in the question. A number without units or context earns no marks.

  \item Every question asks for a short written observation. Code without commentary caps you at half marks.

  \item The lab quiz that follows will ask you to read, trace and modify \emph{your own} submitted code.

\end{itemize}

\textbf{Submit:}

\begin{itemize}[leftmargin=*]

  \item Notebook: \texttt{AML\_LA05\_<SAPID>.ipynb} with all cells executed and outputs visible.

  \item Report: 2--3 pages PDF, or a \texttt{README.md}, carrying your results table and observations.

  \item Do not upload datasets. Cite the source and include the loading code.

\end{itemize}

\section*{Section A (Easy) --- attempt any 2 of 3 \quad (2 $\times$ 2 = 4)}

\begin{enumerate}[label=\textbf{A\arabic*.}, leftmargin=*]

  \item \textbf{Iris in ten lines} \QMarks{2}\\
    \emph{Dataset: Iris (\texttt{load\_iris}).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Split 75/25 stratified, scale, and fit \texttt{KNeighborsClassifier(n\_neighbors=5)}.
      \item Report test accuracy and the confusion matrix.
      \item Name the two species that get confused and state which measurement separates them best.
    \end{enumerate}

  \item \textbf{Digits as vectors} \QMarks{2}\\
    \emph{Dataset: B1 (digits).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Display any five images with their labels using \texttt{imshow}.
      \item State the shape of one image and the shape of the flattened feature matrix.
      \item In two lines, state what information the flattening throws away.
    \end{enumerate}

  \item \textbf{Choosing k} \QMarks{2}\\
    \emph{Dataset: B1 (digits).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Fit KNN for $k \in \{1,3,5,7,9,11,15\}$ and plot test accuracy against $k$.
      \item Report the best $k$.
      \item Explain in two lines what goes wrong at $k=1$ and what goes wrong at a very large $k$.
    \end{enumerate}

\end{enumerate}

\section*{Section B (Medium) --- attempt any 1 of 2 \quad (1 $\times$ 4 = 4)}

\begin{enumerate}[label=\textbf{B\arabic*.}, leftmargin=*]

  \item \textbf{Three classifiers, one dataset} \QMarks{4}\\
    \emph{Dataset: B1 (digits).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Train KNN, an SVM with an RBF kernel and a random forest on the same split.
      \item Report accuracy and macro-F1 for each, plus fit time.
      \item Recommend one for a device with tight memory and explain in 4--6 lines.
    \end{enumerate}

  \item \textbf{Which digits get confused} \QMarks{4}\\
    \emph{Dataset: B1 (digits).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Take your best model and plot the full 10$\times$10 confusion matrix as a heatmap.
      \item Report the three most frequently confused ordered pairs.
      \item Display four misclassified images and explain in 4--6 lines what makes them hard.
    \end{enumerate}

\end{enumerate}

\section*{Section C (Hard) --- compulsory \quad (2)}

\begin{enumerate}[label=\textbf{C\arabic*.}, leftmargin=*]

\end{enumerate}
