  \item Use a train/test split (80/20 unless stated) and set \texttt{random\_state} for reproducibility.

  \item Every question asks for a short written observation. Code without commentary caps you at half marks.

  \item The lab quiz that follows will ask you to read, trace and modify \emph{your own} submitted code.

\section*{Section A (Easy) --- attempt any 2 of 3 \quad (2 $\times$ 2 = 4)}

\begin{enumerate}[label=\textbf{A\arabic*.}, leftmargin=*]

  \item \textbf{Environment proof} \QMarks{2}\\
    \emph{No dataset.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Create a fresh environment and install \texttt{numpy}, \texttt{pandas}, \texttt{scikit-learn}, \texttt{matplotlib}, \texttt{seaborn}, \texttt{jupyter}.
      \item Print the version of each of the five libraries and your Python version in one cell.
      \item State in two lines why pinning versions matters when you hand work to someone else.
    \end{enumerate}

  \item \textbf{First look at the data} \QMarks{2}\\
    \emph{Dataset: A1 (California Housing).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Load with \texttt{fetch\_california\_housing(as\_frame=True)} and run \texttt{head}, \texttt{info}, \texttt{describe} and \texttt{isna().sum()} on \texttt{d.frame}.
      \item State the number of rows, the number of features, the target, and its units. (\texttt{print(d.DESCR)} tells you the units --- the \texttt{return\_X\_y} form throws them away.)
      \item Name one feature you suspect is on a very different scale from the others, with evidence.
    \end{enumerate}

  \item \textbf{Missing values and outliers} \QMarks{2}\\
    \emph{Dataset: A2 (Telco Customer Churn).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Find the column that looks numeric but is stored as text, and explain why.
      \item Convert it, count how many values become missing, and impute them with a stated strategy.
      \item Draw a boxplot of \texttt{tenure} and \texttt{MonthlyCharges} and comment on outliers in two lines.
    \end{enumerate}

\end{enumerate}

\section*{Section B (Medium) --- attempt any 1 of 2 \quad (1 $\times$ 4 = 4)}

\begin{enumerate}[label=\textbf{B\arabic*.}, leftmargin=*]

  \item \textbf{Scaling changes the answer} \QMarks{4}\\
    \emph{Dataset: A1 (California Housing).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Split 80/20 with \texttt{random\_state=0}. Train a \texttt{KNeighborsRegressor(n\_neighbors=5)} on the raw features and report test MAE.
      \item Repeat inside a \texttt{Pipeline} with \texttt{StandardScaler}. Report test MAE again.
      \item Explain in 4--6 lines why the difference appears, referring to how KNN measures distance.
    \end{enumerate}

  \item \textbf{Correlation and redundant features} \QMarks{4}\\
    \emph{Dataset: A2 (Telco Customer Churn).}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Convert \texttt{TotalCharges} with \texttt{pd.to\_numeric(df['TotalCharges'], errors='coerce')}, then encode \texttt{Churn} as 0/1.
      \item Compute \texttt{df.corr(numeric\_only=True)} --- the plain \texttt{df.corr()} raises \texttt{ValueError} on this frame because of the text columns. Rank the four numeric features against \texttt{Churn} and report the strongest three.
      \item \texttt{TotalCharges} is roughly \texttt{tenure}$\times$\texttt{MonthlyCharges}. Explain in 4--6 lines what problem that redundancy causes, and what you would do about it.
    \end{enumerate}

\end{enumerate}
