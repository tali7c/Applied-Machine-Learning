\section*{Objective}

\begin{itemize}[leftmargin=*]

  \item Fit simple and multiple linear regression and read the coefficients.

  \item Evaluate a regressor with metrics that carry the units of the target.

  \item Recognise over-fitting and control it with regularization.

  \item \textbf{Before the lab:} run \texttt{fetch\_california\_housing()} once at home. It downloads \textasciitilde1.4\,MB and the campus network may block it.

\end{itemize}

\textbf{A1 (California Housing).}\\ Full descriptions, sources and loading snippets for all anchor datasets are in \texttt{Anchor-Datasets.pdf}, alongside this brief.

\section*{Instructions}

\begin{itemize}[leftmargin=*]

  \item Use a train/test split (80/20 unless stated) and set \texttt{random\_state} for reproducibility.

  \item Fit every transformer inside a \texttt{Pipeline} so nothing is fitted on the test set.

  \item Report the metric named in the question. A number without units or context earns no marks.

  \item Every question asks for a short written observation. Code without commentary caps you at half marks.

  \item The lab quiz that follows will ask you to read, trace and modify \emph{your own} submitted code.

\end{itemize}

\section*{Section A (Easy) --- attempt any 2 of 3 \quad (2 $\times$ 2 = 4)}

\begin{enumerate}[label=\textbf{A\arabic*.}, leftmargin=*]

  \item \textbf{Simple linear regression} \QMarks{2}\\
    \emph{Dataset: A1. Target: \texttt{MedHouseVal}.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Fit a regression of \texttt{MedHouseVal} on \texttt{MedInc} alone.
      \item Report the slope, the intercept and test $R^2$.
      \item State in one sentence what the slope means in real units.
    \end{enumerate}

  \item \textbf{Multiple linear regression} \QMarks{2}\\
    \emph{Dataset: A1, all features.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Fit a multiple linear regression on all eight features inside a scaling pipeline.
      \item Report test MAE, RMSE and $R^2$.
      \item Plot predicted against actual and describe the pattern in two lines.
    \end{enumerate}

  \item \textbf{Metric choice} \QMarks{2}\\
    \emph{Dataset: A1.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Compute MAE, MSE and RMSE for your model from A2.
      \item State which of the three is hardest to explain to a non-technical client, and why.
      \item State which you would report if a few very expensive districts matter most, and why.
    \end{enumerate}

\end{enumerate}

\section*{Section B (Medium) --- attempt any 1 of 2 \quad (1 $\times$ 4 = 4)}

\begin{enumerate}[label=\textbf{B\arabic*.}, leftmargin=*]

  \item \textbf{Polynomial features and over-fitting} \QMarks{4}\\
    \emph{Dataset: A1.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Build pipelines with \texttt{PolynomialFeatures} of degree 1, 2 and 3 followed by \texttt{LinearRegression}.
      \item Report train and test RMSE for each degree in one table.
      \item Identify the degree at which over-fitting begins and explain in 4--6 lines how the table shows it.
    \end{enumerate}

  \item \textbf{Ridge and Lasso} \QMarks{4}\\
    \emph{Dataset: A1 with degree-2 polynomial features.}
    \begin{enumerate}[label=(\alph*), leftmargin=*]
      \item Fit Ridge and Lasso for $\alpha \in \{0.01, 0.1, 1, 10, 100\}$ and plot test RMSE against $\alpha$.
      \item Report the best $\alpha$ for each and the count of exactly-zero coefficients in the best Lasso.
      \item Explain in 4--6 lines why Lasso zeroes coefficients and Ridge does not.
    \end{enumerate}

\end{enumerate}
